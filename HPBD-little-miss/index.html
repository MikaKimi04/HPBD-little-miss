<script>
const surprises = {
  "CHOCO01": {
    type: "youtube",
    content: "dQw4w9WgXcQ",
    message: "Cette chanson me fait penser à toi ❤️"
  },
  "CHOCO02": {
    type: "text",
    message: "Tu es incroyable, ne l'oublie jamais 💫"
  },
  "CHOCO03": {
    type: "image",
    content: "https://picsum.photos/400",
    message: "Un joli souvenir 📸"
  }
  // 👉 tu continues jusqu'à CHOCO50
};

function checkCode() {
  const code = document.getElementById("codeInput").value.toUpperCase();
  const result = document.getElementById("result");

  if (!surprises[code]) {
    result.innerHTML = "❌ Code inconnu";
    return;
  }

  const surprise = surprises[code];
  let html = `<h2>${surprise.message}</h2>`;

  if (surprise.type === "youtube") {
    html += `<iframe width="300" height="170"
      src="https://www.youtube.com/embed/${surprise.content}"
      frameborder="0" allowfullscreen></iframe>`;
  }

  if (surprise.type === "image") {
    html += `<img src="${surprise.content}" width="300">`;
  }

  result.innerHTML = html;
}
</script>
